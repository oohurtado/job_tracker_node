users
  │
  ├── 1:1 ── personal_information
  │
  └── 1:N ── jobs
                │
                └── 1:N ── job_events
                              │
                              └── 1:N ── interview_questions

/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*

* un "users" 1:1 "personal infrmation"
* un "users" 1:N "jobs"
* un "jobs" 1:N "job_events"
* un "job_events" 1:N "job_questions"

/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*

TABLE - users
- id [PK]
- email : string
- password : string
- first_name : string
- last_name : string
- created_at : datetime

TABLE - personal_information
- id [PK]
- user_id [FK]
- birth_date : datetime
- birth_state : ?string
- birth_country : ?string
- updated_at : datetime
! UNIQUE=user_id

TABLE - jobs
- id [PK]
- user_id [FK]
- company_name : string
- position_name : ?string
- work_model : ?string (remote, hybrid, on_site)
- source_detail : ?string // ...
- source_type : ?string // (url/correo/amigo/llamada/whatsapp/periodico/newsletter)
- details : ?string
- salary_min : number
- salary_max : number
- salary_currency : string (MXN, USD)
- salary_period (hour, month, year)

TABLE - job_events
- id [PK]
- job_id [FK]
- status
- details : string
- event_at : datetime

TABLE - interview_questions
- id [PK]
- job_event_id [FK]
- question : string
- answer : ?string
- notes : ?string
