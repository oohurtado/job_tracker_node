* un "users" 1:N "jobs"
* un "jobs" 1:N "job_events"

/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*/*

TABLE - users
- id [PK]
- email : string
- password : string
- first_name : string
- last_name : string
- created_at : datetime

TABLE - personal_information
- id [PK]
- user_id [FK]
- birth_date : datetime
- birth_state : ?string
- birth_country : ?string
- updated_at : datetime
! UNIQUE=user_id

TABLE - jobs
- id [PK]
- user_id [FK]
- company_name : string
- position_name : ?string
- work_model : ?string (remote, hybrid, on_site)
- source_detail : ?string // ...
- source_type : ?string // (url/correo/amigo/llamada/whatsapp/periodico/newsletter)
- details : ?string
- salary_min : number
- salary_max : number
- salary_currency : string (MXN, USD)
- salary_period (hour, month, year)

TABLE - job_events
- id [PK]
- job_id [FK]
- status
        Guardada | saved | La encontraste pero aún no aplicas
        Postulado | applied | Ya enviaste tu postulación
        En revisión	| in_review	| Confirmaron recepción o la están evaluando
        Contactado | contacted | Un reclutador te escribió o llamó
        En entrevistas | interviewing |	Estás en el proceso (1.ª, 2.ª, 3.ª entrevista)
        Prueba técnica | assessment	| Te pidieron un examen o ejercicio
        Oferta | offer | Te hicieron una oferta
        Aceptada | accepted	| Aceptaste la oferta
        Rechazada | rejected | La empresa te descartó
        Declinada | declined | Tú rechazaste la oferta o retiraste tu candidatura
        Sin respuesta | no_response | Pasó el tiempo y nadie contestó
        Cerrada	| closed | La vacante se canceló o se cubrió
- details : string
- event_at : datetime

TABLE - interview_questions
- id [PK]
- job_event_id [FK]
- question : string
- answer : ?string
- notes : ?string
